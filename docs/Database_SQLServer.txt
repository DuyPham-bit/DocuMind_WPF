IF OBJECT_ID(N'dbo.ProviderSettings', N'U') IS NULL
CREATE TABLE dbo.ProviderSettings (
    Id int IDENTITY(1,1) NOT NULL CONSTRAINT PK_ProviderSettings PRIMARY KEY,
    ProviderType int NOT NULL,
    EncryptedApiKey nvarchar(max) NOT NULL,
    BaseUrl nvarchar(max) NOT NULL,
    ChatModel nvarchar(max) NOT NULL,
    EmbeddingModel nvarchar(max) NOT NULL,
    Temperature float NOT NULL,
    IsActive bit NOT NULL
);
IF OBJECT_ID(N'dbo.Workspaces', N'U') IS NULL
CREATE TABLE dbo.Workspaces (
    Id uniqueidentifier NOT NULL CONSTRAINT PK_Workspaces PRIMARY KEY,
    Name nvarchar(200) NOT NULL,
    Description nvarchar(max) NOT NULL,
    SystemPrompt nvarchar(2000) NOT NULL,
    CreatedAt datetime2 NOT NULL
);
IF OBJECT_ID(N'dbo.ChatMessages', N'U') IS NULL
CREATE TABLE dbo.ChatMessages (
    Id uniqueidentifier NOT NULL CONSTRAINT PK_ChatMessages PRIMARY KEY,
    WorkspaceId uniqueidentifier NOT NULL,
    Role int NOT NULL,
    Content nvarchar(max) NOT NULL,
    SourcesJson nvarchar(max) NULL,
    Timestamp datetime2 NOT NULL,
    CONSTRAINT FK_ChatMessages_Workspaces_WorkspaceId FOREIGN KEY (WorkspaceId)
        REFERENCES dbo.Workspaces(Id) ON DELETE CASCADE
);
IF OBJECT_ID(N'dbo.DocumentChunks', N'U') IS NULL
CREATE TABLE dbo.DocumentChunks (
    Id uniqueidentifier NOT NULL CONSTRAINT PK_DocumentChunks PRIMARY KEY,
    WorkspaceId uniqueidentifier NOT NULL,
    FileName nvarchar(300) NOT NULL,
    PageNumber int NOT NULL,
    Content nvarchar(max) NOT NULL,
    EmbeddingJson nvarchar(max) NOT NULL,
    CreatedAt datetime2 NOT NULL,
    CONSTRAINT FK_DocumentChunks_Workspaces_WorkspaceId FOREIGN KEY (WorkspaceId)
        REFERENCES dbo.Workspaces(Id) ON DELETE CASCADE
);
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.ChatMessages') AND name=N'IX_ChatMessages_WorkspaceId')
CREATE INDEX IX_ChatMessages_WorkspaceId ON dbo.ChatMessages(WorkspaceId);
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.DocumentChunks') AND name=N'IX_DocumentChunks_WorkspaceId')
CREATE INDEX IX_DocumentChunks_WorkspaceId ON dbo.DocumentChunks(WorkspaceId);
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.DocumentChunks') AND name=N'IX_DocumentChunks_WorkspaceId_FileName')
CREATE INDEX IX_DocumentChunks_WorkspaceId_FileName ON dbo.DocumentChunks(WorkspaceId, FileName);
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE object_id=OBJECT_ID(N'dbo.ProviderSettings') AND name=N'IX_ProviderSettings_ProviderType')
CREATE UNIQUE INDEX IX_ProviderSettings_ProviderType ON dbo.ProviderSettings(ProviderType);

-- QUERY DOC DU LIEU
SELECT Id, Name, Description, CreatedAt FROM dbo.Workspaces ORDER BY CreatedAt;
SELECT WorkspaceId, FileName, COUNT(*) AS ChunkCount,
       MIN(PageNumber) AS FirstPage, MAX(PageNumber) AS LastPage
FROM dbo.DocumentChunks GROUP BY WorkspaceId, FileName ORDER BY FileName;
SELECT w.Id, w.Name, COUNT(c.Id) AS ChunkCount, COUNT(DISTINCT c.FileName) AS FileCount
FROM dbo.Workspaces w LEFT JOIN dbo.DocumentChunks c ON c.WorkspaceId=w.Id
GROUP BY w.Id,w.Name;
SELECT TOP (20) FileName, PageNumber, Content, LEN(EmbeddingJson) AS VectorJsonChars
FROM dbo.DocumentChunks WHERE WorkspaceId='11111111-1111-1111-1111-111111111111'
ORDER BY FileName, PageNumber;
-- Khong tra ve ciphertext khi chia se ket qua query.
SELECT ProviderType, ChatModel, EmbeddingModel, BaseUrl, Temperature, IsActive
FROM dbo.ProviderSettings WHERE IsActive=1;
SELECT COUNT(*) AS ActiveProviderCount FROM dbo.ProviderSettings WHERE IsActive=1;
-- Bang nay co the rong: UI hien tai chua luu chat.
SELECT TOP (20) WorkspaceId, Role, Content, Timestamp FROM dbo.ChatMessages ORDER BY Timestamp;
SELECT c.Id AS OrphanChunk FROM dbo.DocumentChunks c
LEFT JOIN dbo.Workspaces w ON w.Id=c.WorkspaceId WHERE w.Id IS NULL;
SELECT fk.name, fk.delete_referential_action_desc FROM sys.foreign_keys fk
WHERE fk.parent_object_id IN (OBJECT_ID(N'dbo.DocumentChunks'),OBJECT_ID(N'dbo.ChatMessages'));
SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, CHARACTER_MAXIMUM_LENGTH, IS_NULLABLE
FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_SCHEMA=N'dbo' AND TABLE_NAME IN
(N'Workspaces',N'DocumentChunks',N'ProviderSettings',N'ChatMessages') ORDER BY TABLE_NAME, ORDINAL_POSITION;
GO
