-- 1. CAU TRUC VA INDEX
BEGIN TRANSACTION;
CREATE TABLE IF NOT EXISTS "ProviderSettings" (
    "Id" INTEGER NOT NULL CONSTRAINT "PK_ProviderSettings" PRIMARY KEY AUTOINCREMENT,
    "ProviderType" INTEGER NOT NULL,
    "EncryptedApiKey" TEXT NOT NULL,
    "BaseUrl" TEXT NOT NULL,
    "ChatModel" TEXT NOT NULL,
    "EmbeddingModel" TEXT NOT NULL,
    "Temperature" REAL NOT NULL,
    "IsActive" INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS "Workspaces" (
    "Id" TEXT NOT NULL CONSTRAINT "PK_Workspaces" PRIMARY KEY,
    "Name" TEXT NOT NULL,
    "Description" TEXT NOT NULL,
    "SystemPrompt" TEXT NOT NULL,
    "CreatedAt" TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS "ChatMessages" (
    "Id" TEXT NOT NULL CONSTRAINT "PK_ChatMessages" PRIMARY KEY,
    "WorkspaceId" TEXT NOT NULL,
    "Role" INTEGER NOT NULL,
    "Content" TEXT NOT NULL,
    "SourcesJson" TEXT NULL,
    "Timestamp" TEXT NOT NULL,
    CONSTRAINT "FK_ChatMessages_Workspaces_WorkspaceId"
      FOREIGN KEY ("WorkspaceId") REFERENCES "Workspaces" ("Id") ON DELETE CASCADE
);
CREATE TABLE IF NOT EXISTS "DocumentChunks" (
    "Id" TEXT NOT NULL CONSTRAINT "PK_DocumentChunks" PRIMARY KEY,
    "WorkspaceId" TEXT NOT NULL,
    "FileName" TEXT NOT NULL,
    "PageNumber" INTEGER NOT NULL,
    "Content" TEXT NOT NULL,
    "EmbeddingJson" TEXT NOT NULL,
    "CreatedAt" TEXT NOT NULL,
    CONSTRAINT "FK_DocumentChunks_Workspaces_WorkspaceId"
      FOREIGN KEY ("WorkspaceId") REFERENCES "Workspaces" ("Id") ON DELETE CASCADE
);
CREATE INDEX IF NOT EXISTS "IX_ChatMessages_WorkspaceId" ON "ChatMessages" ("WorkspaceId");
CREATE INDEX IF NOT EXISTS "IX_DocumentChunks_WorkspaceId" ON "DocumentChunks" ("WorkspaceId");
CREATE INDEX IF NOT EXISTS "IX_DocumentChunks_WorkspaceId_FileName" ON "DocumentChunks" ("WorkspaceId", "FileName");
CREATE UNIQUE INDEX IF NOT EXISTS "IX_ProviderSettings_ProviderType" ON "ProviderSettings" ("ProviderType");

-- 3. QUERY DOC DU LIEU
SELECT "Id", "Name", "Description", "CreatedAt" FROM "Workspaces" ORDER BY "CreatedAt";
SELECT "WorkspaceId", "FileName", COUNT(*) AS "ChunkCount",
       MIN("PageNumber") AS "FirstPage", MAX("PageNumber") AS "LastPage"
FROM "DocumentChunks" GROUP BY "WorkspaceId", "FileName" ORDER BY "FileName";
SELECT w."Name", COUNT(c."Id") AS "ChunkCount", COUNT(DISTINCT c."FileName") AS "FileCount"
FROM "Workspaces" w LEFT JOIN "DocumentChunks" c ON c."WorkspaceId" = w."Id"
GROUP BY w."Id", w."Name";
SELECT "FileName", "PageNumber", "Content", length("EmbeddingJson") AS "VectorJsonChars"
FROM "DocumentChunks" WHERE "WorkspaceId" = '11111111-1111-1111-1111-111111111111'
ORDER BY "FileName", "PageNumber" LIMIT 20;
-- Chi hien metadata; KHONG SELECT EncryptedApiKey de chia se ra ngoai.
SELECT "ProviderType", "ChatModel", "EmbeddingModel", "BaseUrl", "Temperature", "IsActive"
FROM "ProviderSettings" WHERE "IsActive" = 1;
SELECT COUNT(*) AS "ActiveProviderCount" FROM "ProviderSettings" WHERE "IsActive" = 1;
-- ChatMessages rong la dung voi UI hien tai: app chua luu chat xuong DB.
SELECT "WorkspaceId", "Role", "Content", "Timestamp" FROM "ChatMessages" ORDER BY "Timestamp" LIMIT 20;
SELECT c."Id" AS "OrphanChunk" FROM "DocumentChunks" c
LEFT JOIN "Workspaces" w ON w."Id" = c."WorkspaceId" WHERE w."Id" IS NULL;

PRAGMA index_list('DocumentChunks');
